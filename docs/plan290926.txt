   ```prisma
     FINISH_TO_START // Padrão: A predecessora precisa terminar para a sucessora iniciar

     dependencyType   DependencyType @default(FINISH_TO_START)

     @@unique([taskId, dependsOnTaskId])
     @@index([taskId])
     @@index([dependsOnTaskId])
   }
   ```

   ```typescript
   const filteredTasks = useMemo(() => {
     return tasks.filter(task => {
       // 1. Escopo (Root vs Subtasks)
       if (filters.scope === 'root_only' && task.parentTaskId) return false;
       if (filters.scope === 'subtasks_only' && !task.parentTaskId) return false;

       // 2. Responsável
       if (filters.assigneeId && task.assigneeId !== filters.assigneeId) return false;

       // 3. Status e Prioridade
       if (filters.statuses.length > 0 && !filters.statuses.includes(task.status)) return false;
       if (filters.priorities.length > 0 && !filters.priorities.includes(task.priority)) return false;

       // 4. Múltiplas Tags
       if (filters.tagIds.length > 0) {
         const taskTags = getTaskTags(task.id);
         const hasAnyTag = taskTags.some(t => filters.tagIds.includes(t.id));
         if (!hasAnyTag) return false;
       }

       // 5. Busca Difusa Levenshtein (tolerância a erros)
       if (searchQuery.trim()) {
         return searchIndexService.matches(task, searchQuery);
       }

       return true;
     });
   }, [tasks, filters, searchQuery]);
   ```
